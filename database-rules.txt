Due Date: rule for the Chop the Check Realtime Database (Firebase project "chopthecheck").
Paste the block below inside "rules": { ... }, next to the other entries (e.g. "rooms"), with a comma between entries.

    "dueDate": {
      ".read": true,
      "people": {
        "$id": {
          ".write": true,
          ".validate": "newData.hasChildren(['name', 'phone', 'at']) && newData.child('name').isString() && newData.child('name').val().length >= 1 && newData.child('name').val().length <= 24 && newData.child('phone').isString() && newData.child('phone').val().length <= 40 && newData.child('at').isNumber()"
        }
      },
      "aimee": {
        ".write": true,
        ".validate": "newData.hasChildren(['pid', 'phone']) && newData.child('pid').isString() && newData.child('phone').isString()"
      },
      "due": { ".write": true, ".validate": "newData.isString() && newData.val().matches(/^[0-9]{4}-[0-9]{2}-[0-9]{2}$/)" },
      "birth": { ".write": true, ".validate": "newData.isString() && newData.val().matches(/^[0-9]{4}-[0-9]{2}-[0-9]{2}$/)" },
      "locked": { ".write": true, ".validate": "newData.isBoolean()" },
      "amount": { ".write": true, ".validate": "newData.isNumber() && newData.val() > 0 && newData.val() <= 1000" },
      "order": {
        ".write": true,
        "$n": { ".validate": "newData.isString() && newData.val().length <= 40" }
      },
      "dates": {
        "$day": {
          ".write": "!data.exists() || !newData.exists()",
          ".validate": "$day.matches(/^[0-9]{4}-[0-9]{2}-[0-9]{2}$/) && newData.isString() && newData.val().length <= 40"
        }
      },
      "skipped": {
        "$id": { ".write": true, ".validate": "newData.isBoolean()" }
      },
      "$other": { ".validate": false }
    }

What it does: anyone with the link can read the pool and add names. A date can only be
taken once (it can't be overwritten, only cleared). Anything else under dueDate is refused.
